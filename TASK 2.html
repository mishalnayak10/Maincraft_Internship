# Step 1: Import Required Libraries[cite: 3]
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import fetch_california_housing
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression, Ridge
from sklearn.tree import DecisionTreeRegressor
from sklearn.metrics import mean_squared_error, r2_score

# Step 2: Load the Dataset[cite: 3]
data = fetch_california_housing(as_frame=True)
df = pd.concat([data.data, data.target.rename("HousePrice")], axis=1)

# Display the first few rows to review column names and data types[cite: 3]
display(df.head())

# Step 3: Separate Features and Target Variable[cite: 3]
X = df.drop("HousePrice", axis=1)
y = df["HousePrice"]

# Step 4: Feature Scaling (Critical Step)[cite: 3]
# After scaling, all features have a common scale, improving learning stability and performance[cite: 3].
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# Step 5: Train-Test Split[cite: 3]
# This ensures the model is evaluated using unseen data rather than memorized values[cite: 3].
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, random_state=42
)

# Step 6: Train Multiple Models[cite: 3]
models = {
    "Linear Regression": LinearRegression(),
    "Ridge Regression": Ridge(alpha=1.0),
    "Decision Tree": DecisionTreeRegressor(max_depth=5)
}

# Step 7: Model Evaluation and Comparison[cite: 3]
results = {}
for name, model in models.items():
    model.fit(X_train, y_train)
    predictions = model.predict(X_test)
    
    # Calculate RMSE using np.sqrt to avoid the TypeError in newer scikit-learn versions
    rmse = np.sqrt(mean_squared_error(y_test, predictions))
    r2 = r2_score(y_test, predictions)
    
    results[name] = {
        "RMSE": rmse,
        "R2 Score": r2
    }

# Create a model performance comparison table[cite: 3]
results_df = pd.DataFrame(results).T
print("Model Performance Comparison:")
display(results_df)

# Step 8: Visual Performance Validation[cite: 3]
best_model = LinearRegression()
best_model.fit(X_train, y_train)
y_pred = best_model.predict(X_test)

plt.figure(figsize=(6,6))
plt.scatter(y_test, y_pred, alpha=0.4)
plt.xlabel("Actual House Prices")
plt.ylabel("Predicted House Prices")
plt.title("Actual vs Predicted House Prices")
plt.plot([y_test.min(), y_test.max()],
         [y_test.min(), y_test.max()],
         color="red")
plt.show()
